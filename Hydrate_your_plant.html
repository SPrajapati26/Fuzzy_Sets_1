import numpy as np
import skfuzzy as fuzz
from skfuzzy import control as ctrl
import matplotlib.pyplot as plt

# Inputs
soil_moisture = ctrl.Antecedent(
    np.arange(0, 101, 1),
    'soil_moisture'
)

temperature = ctrl.Antecedent(
    np.arange(0, 41, 1),
    'temperature'
)

sunlight = ctrl.Antecedent(
    np.arange(0, 11, 1),
    'sunlight'
)

# Output
water_amount = ctrl.Consequent(
    np.arange(0, 101, 1),
    'water_amount'
)

rule1 = ctrl.Rule(
    soil_moisture['wet'],
    water_amount['none']
)

rule2 = ctrl.Rule(
    soil_moisture['moist'] & temperature['low'],
    water_amount['low']
)

rule3 = ctrl.Rule(
    soil_moisture['moist'] & temperature['medium'],
    water_amount['medium']
)

rule4 = ctrl.Rule(
    soil_moisture['moist'] & temperature['high'],
    water_amount['high']
)

rule5 = ctrl.Rule(
    soil_moisture['dry'] & temperature['low'],
    water_amount['medium']
)

rule6 = ctrl.Rule(
    soil_moisture['dry'] & temperature['medium'],
    water_amount['high']
)

rule7 = ctrl.Rule(
    soil_moisture['dry'] & temperature['high'],
    water_amount['very_high']
)

rule8 = ctrl.Rule(
    soil_moisture['dry'] & sunlight['high'],
    water_amount['very_high']
)

rule9 = ctrl.Rule(
    soil_moisture['dry'] & sunlight['medium'],
    water_amount['high']
)

rule10 = ctrl.Rule(
    soil_moisture['moist'] & sunlight['high'],
    water_amount['high']
)

rule11 = ctrl.Rule(
    soil_moisture['wet'] & temperature['high'],
    water_amount['low']
)

rule12 = ctrl.Rule(
    soil_moisture['wet'] & sunlight['high'],
    water_amount['low']
)


plant_ctrl = ctrl.ControlSystem([
    rule1, rule2, rule3, rule4,
    rule5, rule6, rule7, rule8,
    rule9, rule10, rule11, rule12
])

plant = ctrl.ControlSystemSimulation(plant_ctrl)

# Generate trapezoidal membership function on range [0, 1]
x = np.arange(0, 101, 1)

#fuzzy output
mfx = np.fmax(
    water_amount['high'].mf,
    water_amount['very_high'].mf
)

defuzz_centroid = fuzz.defuzz(x, mfx, 'centroid')
defuzz_bisector = fuzz.defuzz(x, mfx, 'bisector')
defuzz_mom = fuzz.defuzz(x, mfx, 'mom')
defuzz_som = fuzz.defuzz(x, mfx, 'som')
defuzz_lom = fuzz.defuzz(x, mfx, 'lom')

print("Centroid:", defuzz_centroid)
print("Bisector:", defuzz_bisector)
print("Mean of Maximum:", defuzz_mom)
print("Smallest of Maximum:", defuzz_som)
print("Largest of Maximum:", defuzz_lom)

labels = [
    'Centroid',
    'Bisector',
    'Mean of Maximum',
    'Smallest of Maximum',
    'Largest of Maximum'
]

xvals = [
    defuzz_centroid,
    defuzz_bisector,
    defuzz_mom,
    defuzz_som,
    defuzz_lom
]

ymax = [
    fuzz.interp_membership(x, mfx, i)
    for i in xvals
]

plt.figure(figsize=(10, 6))

plt.plot(x, mfx, 'k')

for xv, y, label in zip(xvals, ymax, labels):
    plt.vlines(
        xv,
        0,
        y,
        label=f'{label}: {xv:.2f}'
    )

plt.xlabel('Watering Requirement (%)')
plt.ylabel('Fuzzy Membership')
plt.title('Defuzzification Methods - Smart Plant Watering')

plt.legend()
plt.show()
